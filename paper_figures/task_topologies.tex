\begin{figure}[tb]
    \centering
    \begin{tikzpicture}[
        >=Stealth,
        vertex/.style={draw, circle, minimum size=0.9cm, inner sep=0pt}
    ]
        \node[vertex] (ts) at (0,1.5)   {$Ts$};
        \node[vertex] (a)  at (2.6,2.5) {$A$};
        \node[vertex] (b)  at (2.6,0.5) {$B$};
        \node[vertex] (a1) at (5.2,3)   {$A1$};
        \node[vertex] (a2) at (5.2,2)   {$A2$};
        \node[vertex] (b1) at (5.2,1)   {$B1$};
        \node[vertex] (b2) at (5.2,0)   {$B2$};
        \node[vertex] (tm) at (7.8,1.5) {$Tm$};

        \draw[->, line width=1pt] ($(ts)!0.45cm!(a)$)  -- ($(a)!0.45cm!(ts)$);
        \draw[->, line width=1pt] ($(ts)!0.45cm!(b)$)  -- ($(b)!0.45cm!(ts)$);
        \draw[->, line width=1pt] ($(a)!0.45cm!(a1)$)  -- ($(a1)!0.45cm!(a)$);
        \draw[->, line width=1pt] ($(a)!0.45cm!(a2)$)  -- ($(a2)!0.45cm!(a)$);
        \draw[->, line width=1pt] ($(b)!0.45cm!(b1)$)  -- ($(b1)!0.45cm!(b)$);
        \draw[->, line width=1pt] ($(b)!0.45cm!(b2)$)  -- ($(b2)!0.45cm!(b)$);
        \draw[->, line width=1pt] ($(a1)!0.45cm!(tm)$) -- ($(tm)!0.45cm!(a1)$);
        \draw[->, line width=1pt] ($(a2)!0.45cm!(tm)$) -- ($(tm)!0.45cm!(a2)$);
        \draw[->, line width=1pt] ($(b1)!0.45cm!(tm)$) -- ($(tm)!0.45cm!(b1)$);
        \draw[->, line width=1pt] ($(b2)!0.45cm!(tm)$) -- ($(tm)!0.45cm!(b2)$);
    \end{tikzpicture}
    \caption{DAG topology of $\tau_{\mathrm{HIGH}}$: two nested levels of forking
    ($Ts \rightarrow \{A,B\}$, $A \rightarrow \{A1,A2\}$, $B \rightarrow \{B1,B2\}$)
    converging at a single 4-fan-in sink $Tm$.}
    \label{fig:high-task-topology}
\end{figure}
